    1. Saving Throws (Спасброски)¶

    Текущее состояние: ❌ Модель есть (SavingThrow), выполнение отсутствует

    Проблема: Заклинания Fireball, Hold Person не работают корректно — цели не делают спасброски.

    Требование:

    При применении заклинания с savingThrow != null: 1. Каждая цель делает бросок d20 + ability_modifier + (proficiency_bonus если есть proficiency в saving throw). 2. Если результат >= dc → успех (эффект уменьшается или отменяется). 3. Если результат < dc → провал (полный эффект).

    Типы эффектов: - SaveEffect.HALF — при успехе урон уменьшается вдвое (Fireball) - SaveEffect.NEGATES — при успехе эффект полностью отменяется (Hold Person) - SaveEffect.NONE — спасбросок не влияет (редко)

    Где реализовать:

    Создать новый файл: backend/app/src/main/kotlin/com/yorva/rules/SavingThrowResolver.kt

    package com.yorva.rules

import com.yorva.spells.SavingThrow
import kotlin.random.Random

enum class SaveEffect {
    HALF,      // Урон вдвое при успехе
    NEGATES,   // Эффект отменяется при успехе
    NONE,      // Спасбросок не влияет
}

data class SaveResult(
    val success: Boolean,
    val roll: DiceRoll,
)

class SavingThrowResolver(private val rng: Random = Random.Default) {

    /**
     * Выполняет спасбросок для персонажа.
     */
    fun resolvePlayerSave(
        savingThrow: SavingThrow,
        character: CharacterState,
    ): SaveResult {
        val ability = when (savingThrow.ability) {
            Ability.STR -> character.effectiveAttributes.strength
            Ability.DEX -> character.effectiveAttributes.dexterity
            Ability.CON -> character.effectiveAttributes.constitution
            Ability.INT -> character.effectiveAttributes.intelligence
            Ability.WIS -> character.effectiveAttributes.wisdom
            Ability.CHA -> character.effectiveAttributes.charisma
        }
        val modifier = abilityModifier(ability)
        val proficiencyBonus = if (savingThrow.ability.name in character.proficiencies) {
            proficiencyBonus(character.level)
        } else 0

        val d20 = rng.nextInt(1, 21)
        val total = d20 + modifier + proficiencyBonus
        val success = total >= savingThrow.dc

        val roll = DiceRoll(
            value = total,
            label = "d20+$modifier${if (proficiencyBonus > 0) "+$proficiencyBonus" else ""} (${savingThrow.ability.name} save)",
            isCritical = d20 == 20,
        )

        return SaveResult(success, roll)
    }

    /**
     * Выполняет спасбросок для врага.
     */
    fun resolveEnemySave(
        savingThrow: SavingThrow,
        enemy: Enemy,
    ): SaveResult {
        // Для простоты: враги имеют фиксированный бонус к спасброскам
        val saveBonus = enemy.attackBonus // Упрощение: используем attackBonus как универсальный
        val d20 = rng.nextInt(1, 21)
        val total = d20 + saveBonus
        val success = total >= savingThrow.dc

        val roll = DiceRoll(
            value = total,
            label = "d20+$saveBonus (${savingThrow.ability.name} save)",
            isCritical = d20 == 20,
        )

        return SaveResult(success, roll)
    }

    private fun abilityModifier(score: Int): Int = (score - 10) / 2

    private fun proficiencyBonus(level: Int): Int = 2 + (level - 1) / 4
}

enum class Ability {
    STR, DEX, CON, INT, WIS, CHA
}

private fun resolveAttack(context: ActionContext): TurnResult {
    val character = context.state.character
    val enemy = context.state.combat.enemy

    val strMod = abilityModifier(character.effectiveAttributes.strength)
    val profBonus = proficiencyBonus(character.level)

    val d20 = rng.nextInt(1, 21)

    // ⬇️ ДОБАВИТЬ ПРИМЕНЕНИЕ МОДИФИКАТОРОВ
    val attackModifiers = character.activeModifiers.filter { it.stat == "attack_roll" }

        when (it.type) {
            ModifierType.FLAT -> it.value
            ModifierType.DICE -> rng.nextInt(1, it.value + 1)  // Для 1d4: value=4

    val isCrit = d20 == 20

    val diceRolls = mutableListOf(

            value = attackRoll,
            label = "d20+$strMod+$profBonus${if (modifierBonus > 0) "+$modifierBonus" else ""} (attack)",
            isCritical = isCrit,

    // После выполнения хода игрока:
private fun advanceRound(state: GameState): GameState {
    var character = state.character
    var combat = state.combat

    // ⬇️ УМЕНЬШИТЬ ДЛИТЕЛЬНОСТЬ БАФФОВ/ДЕБАФФОВ
    character = character.copy(
        activeModifiers = character.activeModifiers.mapNotNull { modifier ->
            val newDuration = modifier.durationRounds - 1
            if (newDuration > 0) {
                modifier.copy(durationRounds = newDuration)
            } else {
                null  // Удалить истёкший модификатор
            }
        }
    )

    // ⬇️ УМЕНЬШИТЬ ДЛИТЕЛЬНОСТЬ УСЛОВИЙ ВРАГА
    // (Требуется добавить поле duration в Enemy.conditions)

    return state.copy(
        character = character,
        combat = combat.copy(roundNumber = combat.roundNumber + 1)
    )
}

    4. Enemy AI (Тактика врагов)¶

    Текущее состояние: ⚠️ Враг всегда атакует (примитивно)

    Требование: Враги должны иметь простую тактику и использовать способности.

    Модель:

    enum class EnemyBehavior {
    AGGRESSIVE,   // Всегда атакует
    DEFENSIVE,    // Dodge при HP < 50%
    TACTICAL,     // Использует способности
}

data class EnemyAbility(
    val id: String,
    val name: String,
    val type: AbilityType,        // ATTACK, SPELL, SPECIAL
    val cooldown: Int,            // Раундов до повторного использования
    val lastUsedRound: Int? = null,
)

data class Enemy(
    // ... существующие поля
    val behavior: EnemyBehavior = EnemyBehavior.AGGRESSIVE,
    val abilities: List<EnemyAbility> = emptyList(),
)

    Логика:

    // В TurnResolver.resolveEnemyTurn():
private fun resolveEnemyTurn(context: ActionContext): List<StateChange> {
    val enemy = context.state.combat.enemy
    val character = context.state.character
    val roundNumber = context.state.combat.roundNumber

    when (enemy.behavior) {
        EnemyBehavior.AGGRESSIVE -> {
            // Всегда атакует
            return resolveEnemyAttack(context)
        }

        EnemyBehavior.DEFENSIVE -> {
            // Dodge при HP < 50%
            if (enemy.hp < enemy.maxHp / 2) {
                return listOf(StateChange.ApplyStatus("dodging", Target.ENEMY))
            } else {
                return resolveEnemyAttack(context)
            }
        }

        EnemyBehavior.TACTICAL -> {
            // Использовать способность если доступна
            val availableAbility = enemy.abilities.firstOrNull {
                val lastUsed = it.lastUsedRound ?: -999
                (roundNumber - lastUsed) >= it.cooldown
            }

            if (availableAbility != null) {
                return resolveEnemyAbility(context, availableAbility)
            } else {
                return resolveEnemyAttack(context)
            }
        }
    }
}

    5. Преимущество из скрытности¶

    Текущее состояние: ⚠️ Статус hidden есть, но advantage не автоматический

    Требование: При атаке из hidden → автоматически advantage.

    Логика:

    // В TurnResolver.resolveAttack():
private fun resolveAttack(context: ActionContext): TurnResult {
    val character = context.state.character

    // ⬇️ ПРОВЕРИТЬ СКРЫТНОСТЬ
    val hasAdvantage = "hidden" in character.statuses

    val d20 = if (hasAdvantage) {
        val roll1 = rng.nextInt(1, 21)
        val roll2 = rng.nextInt(1, 21)
        maxOf(roll1, roll2)
    } else {
        rng.nextInt(1, 21)
    }

    // ... остальная логика

    // ⬇️ СНЯТЬ СКРЫТНОСТЬ ПОСЛЕ АТАКИ
    val newStatuses = character.statuses - "hidden"

    return TurnResult(
        // ...
        stateChanges = stateChanges + StateChange.RemoveStatus("hidden", Target.PLAYER)
    )
}

    Приоритет: 🟡 Важно
    Оценка: 1 день

    6. Концентрация при уроне¶

    Текущее состояние: ⚠️ Концентрация НЕ прерывается при уроне

    Требование: При уроне → Constitution saving throw, DC = max(10, damage/2).

    Логика:

    // В TurnResolver, после применения урона:
private fun checkConcentration(
    character: CharacterState,
    damage: Int,
    rng: Random
): StateChange? {
    if (character.concentrationSpell == null) return null

    val dc = maxOf(10, damage / 2)
    val conMod = abilityModifier(character.effectiveAttributes.constitution)
    val profBonus = if ("CON" in character.proficiencies) {
        proficiencyBonus(character.level)
    } else 0

    val d20 = rng.nextInt(1, 21)
    val total = d20 + conMod + profBonus

    return if (total < dc) {
        // Концентрация прервана
        StateChange.BreakConcentration
    } else {
        null  // Концентрация сохранена
    }
}

    Приоритет: 🟡 Важно
    Оценка: 2 дня

    7. Навыки (Skills)¶

    Текущее состояние: ❌ Поле proficiencies есть, но не используется

    Требование: При проверке навыка → добавлять proficiency bonus.

    Логика:

    // В TurnResolver.resolveSkillCheck():
private fun resolveSkillCheck(
    skill: String,
    character: CharacterState,
    dc: Int,
    rng: Random
): DiceRoll {
    val ability = skillToAbility(skill)  // "stealth" -> DEX
    val abilityScore = when (ability) {
        Ability.STR -> character.effectiveAttributes.strength
        Ability.DEX -> character.effectiveAttributes.dexterity
        // ... остальные
    }
    val modifier = abilityModifier(abilityScore)

    val profBonus = if (skill in character.proficiencies) {
        proficiencyBonus(character.level)
    } else 0

    val d20 = rng.nextInt(1, 21)
    val total = d20 + modifier + profBonus

    return DiceRoll(
        value = total,
        label = "d20+$modifier${if (profBonus > 0) "+$profBonus" else ""} ($skill)",
    )
}

private fun skillToAbility(skill: String): Ability = when (skill.lowercase()) {
    "athletics" -> Ability.STR
    "acrobatics", "stealth", "sleight_of_hand" -> Ability.DEX
    "arcana", "history", "investigation", "nature", "religion" -> Ability.INT
    "animal_handling", "insight", "medicine", "perception", "survival" -> Ability.WIS
    "deception", "intimidation", "performance", "persuasion" -> Ability.CHA
    else -> Ability.STR
}

    Оценка: 3 дня
